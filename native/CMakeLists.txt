cmake_minimum_required(VERSION 3.20)
project(sistema_core LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
find_package(SQLite3 REQUIRED)
find_package(LibXml2 REQUIRED)
find_path(ARCHIVE_INCLUDE_DIR archive.h HINTS /opt/homebrew/opt/libarchive/include /usr/local/opt/libarchive/include REQUIRED)
find_library(ARCHIVE_LIBRARY archive HINTS /opt/homebrew/opt/libarchive/lib /usr/local/opt/libarchive/lib REQUIRED)
add_library(sistema_core STATIC scheduler_core.c whisper_mcp_bridge.c metrics_core.c)
target_link_libraries(sistema_core PUBLIC m)
target_compile_options(sistema_core PRIVATE -Wall -Wextra)
add_executable(db_init db_core.c)
target_link_libraries(db_init PRIVATE SQLite::SQLite3)
add_executable(sistema_service service.c domains.c generators.c vendor/cJSON.c vendor/sqlite-vec.c)
target_include_directories(sistema_service PRIVATE ${ARCHIVE_INCLUDE_DIR})
target_link_libraries(sistema_service PRIVATE sistema_core SQLite::SQLite3 LibXml2::LibXml2 ${ARCHIVE_LIBRARY} "-framework CoreFoundation" "-framework CoreGraphics" "-framework CoreText")
set_source_files_properties(vendor/sqlite-vec.c PROPERTIES COMPILE_OPTIONS "-Wno-unused-parameter;-Wno-unused-variable")
target_compile_definitions(sistema_service PRIVATE SQLITE_CORE SQLITE_VEC_STATIC)
target_compile_options(sistema_service PRIVATE -Wall -Wextra)
add_executable(core_test ../tests/core_test.c)
target_link_libraries(core_test PRIVATE sistema_core)
enable_testing()
add_test(NAME core_test COMMAND core_test)
option(WITH_FACE "Build OpenCV ArcFace extractor" OFF)
if(WITH_FACE)
 find_package(OpenCV 4 REQUIRED COMPONENTS core imgcodecs imgproc dnn objdetect)
 add_executable(face_extract face_bridge.cpp)
 target_link_libraries(face_extract PRIVATE ${OpenCV_LIBS})
endif()
option(WITH_WHISPER "Build direct whisper C bridge" OFF)
if(WITH_WHISPER)
 set(WHISPER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/whisper.cpp" CACHE PATH "whisper.cpp checkout")
 set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 add_subdirectory(${WHISPER_ROOT} whisper)
 target_compile_definitions(sistema_core PRIVATE WITH_WHISPER)
 target_link_libraries(sistema_core PUBLIC whisper)
 add_executable(sistema_transcribe transcribe_main.c)
 target_link_libraries(sistema_transcribe PRIVATE sistema_core)
 set_target_properties(sistema_transcribe PROPERTIES LINKER_LANGUAGE CXX)
endif()
# Keep assertion-based tests active in Release builds.
target_compile_options(core_test PRIVATE -UNDEBUG)
